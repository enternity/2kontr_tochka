## Вариант 15 (массивы)

### Условие
Имеется массив из ID пользователей. Используя алгоритм смещения на N разрядов влево (<<) зашифровать элементы массива. Выполнить расшифровку элементов массива с помощью операции >>.

### Код программы

```cpp
#include <iostream>
using namespace std;

int main()
{
    int n;

    cout << "Введите количество ID: ";
    cin >> n;

    int a[100];
    int b[100];
    int c[100];

    cout << "Введите ID:\n";

    for (int i = 0; i < n; i++)
    {
        cin >> a[i];
    }

    int k;

    cout << "Введите количество сдвигов: ";
    cin >> k;

    // Шифрование
    for (int i = 0; i < n; i++)
    {
        b[i] = a[i] << k;
    }

    // Расшифровка
    for (int i = 0; i < n; i++)
    {
        c[i] = b[i] >> k;
    }

    cout << "\nИсходный массив:\n";

    for (int i = 0; i < n; i++)
    {
        cout << a[i] << " ";
    }

    cout << "\n\nЗашифрованный массив:\n";

    for (int i = 0; i < n; i++)
    {
        cout << b[i] << " ";
    }

    cout << "\n\nРасшифрованный массив:\n";

    for (int i = 0; i < n; i++)
    {
        cout << c[i] << " ";
    }

    return 0;
}
Тест программы
